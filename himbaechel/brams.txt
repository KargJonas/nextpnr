# The SLG47910's block RAM for Yosys's memory_libmap (flow.sh): eight 4 kb
# slices, each a write port and a read port, both clocked on either edge,
# with no initial contents (the image cannot load them: the SLG47910's MCU
# stream has no BRAM section). 512 x 8 only: the narrower ratios (RATIO 1-3,
# the word's place in a byte on WDATA) fail on the Shrike when the address
# changes every cycle (bram_map.v).
ram block $__SLG47910_BRAM_ {
	abits 9;
	widths 8 global;
	cost 64;
	init none;
	port sw "W" {
		clock anyedge;
	}
	port sr "R" {
		clock anyedge;
		rden;
	}
}
